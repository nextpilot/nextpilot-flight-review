# 内置业务组件：只在这里聚合，base 不认识具体组件和格式。
# 每个领域一个开关；关掉后对应 target 不参与构建。
if(PS_BUILTIN_AUTOPILOT AND PS_MODULE_DATASOURCE AND PS_MODULE_ANALYSIS
   AND TARGET ps_common_datasources_component)
  add_subdirectory(autopilot)
endif()
if(PS_BUILTIN_AUTOMOTIVE)
  add_subdirectory(automotive)
endif()
if(PS_BUILTIN_ROBOTICS)
  add_subdirectory(robotics)
endif()

if(PS_BUILD_GNSS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/gnss/CMakeLists.txt)
  add_subdirectory(gnss)
endif()

# 聚合 target：base 只链接它；为空时不影响底座构建。
# 组件对象通过 INTERFACE_SOURCES 进入宿主，保证静态注册器不被裁掉。
add_library(ps_builtin_components INTERFACE)

if(TARGET ps_common_datasources_component)
  target_sources(ps_builtin_components INTERFACE $<TARGET_OBJECTS:ps_common_datasources_component>)
  target_link_libraries(ps_builtin_components INTERFACE ps_common_datasources_component)
endif()
if(TARGET ps_application_scripting)
  target_sources(ps_builtin_components INTERFACE $<TARGET_OBJECTS:ps_application_scripting>)
  target_link_libraries(ps_builtin_components INTERFACE ps_application_scripting)
endif()
if(TARGET ps_autopilot_component)
  target_sources(ps_builtin_components INTERFACE $<TARGET_OBJECTS:ps_autopilot_component>)
  target_link_libraries(ps_builtin_components INTERFACE ps_autopilot_component ps_autopilot)
endif()
if(TARGET ps_automotive_component)
  target_sources(ps_builtin_components INTERFACE $<TARGET_OBJECTS:ps_automotive_component>)
  target_link_libraries(ps_builtin_components INTERFACE ps_automotive_component)
endif()
if(TARGET ps_robotics_component)
  target_sources(ps_builtin_components INTERFACE $<TARGET_OBJECTS:ps_robotics_component>)
  target_link_libraries(ps_builtin_components INTERFACE ps_robotics_component)
endif()
if(TARGET ps_gnss_component)
  target_sources(ps_builtin_components INTERFACE $<TARGET_OBJECTS:ps_gnss_component>)
  target_link_libraries(ps_builtin_components INTERFACE ps_gnss_component)
endif()
